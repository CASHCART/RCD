#include <stdio.h>
#include <ctype.h>

int main() {
    char expr[100];
    int i = 0, temp = 1;

    printf("Enter expression: ");
    scanf("%99s", expr);

    printf("\nThree Address Code:\n");

    while (expr[i] != '\0') {
        if (expr[i] == '*' || expr[i] == '/') {
            char left = expr[i - 1];
            char op = expr[i];
            char right = expr[i + 1];

            printf("t%d = %c %c %c\n",
                   temp, left, op, right);

            expr[i - 1] = '0' + temp;
            expr[i] = '\0';
            expr[i + 1] = '\0';

            temp++;
        }
        i++;
    }

    i = 0;

    while (expr[i] != '\0') {
        if (expr[i] == '+' || expr[i] == '-') {
            char left = expr[i - 1];
            char op = expr[i];
            char right = expr[i + 1];

            printf("t%d = %c %c %c\n",
                   temp, left, op, right);

            temp++;
        }
        i++;
    }

    return 0;
}
