#include <stdio.h>
#include <string.h>

int main() {
    int n, i;
    char line[10][100];
    char lhs[20], a[20], b[20], op;

    printf("Enter number of instructions: ");
    scanf("%d", &n);
    getchar();

    printf("Enter Three Address Code:\n");

    for (i = 0; i < n; i++) {
        fgets(line[i], sizeof(line[i]), stdin);
    }

    printf("\nAssembly Code:\n");

    for (i = 0; i < n; i++) {
        if (sscanf(line[i], "%19s = %19s %c %19s",
                   lhs, a, &op, b) == 4) {
            printf("MOV R0, %s\n", a);

            switch (op) {
                case '+': printf("ADD R0, %s\n", b); break;
                case '-': printf("SUB R0, %s\n", b); break;
                case '*': printf("MUL R0, %s\n", b); break;
                case '/': printf("DIV R0, %s\n", b); break;
            }

            printf("MOV %s, R0\n", lhs);
        }
        else if (sscanf(line[i], "%19s = %19s", lhs, a) == 2) {
            printf("MOV %s, %s\n", lhs, a);
        }
    }

    return 0;
}
