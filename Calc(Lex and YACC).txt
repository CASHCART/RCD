calc.l

%{
#include <stdio.h>
#include <stdlib.h>
#include "y.tab.h"

extern int yylval;
%}

%%
[0-9]+     { yylval = atoi(yytext); return NUMBER; }
[+\-*/()]  { return yytext[0]; }
\n         { return '\n'; }
[ \t]      ;
.          { printf("Invalid Character: %s\n", yytext); }
%%

int yywrap() {
    return 1;
}


calc.y
%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *s);
%}

%token NUMBER
%left '+' '-'
%left '*' '/'
%left UMINUS

%%
input:
    /* empty */
    | input line
    ;

line:
    '\n'
    | expr '\n' {
        printf("= %d\n", $1);
    }
    ;

expr:
    NUMBER {
        $$ = $1;
    }
    | expr '+' expr {
        $$ = $1 + $3;
    }
    | expr '-' expr {
        $$ = $1 - $3;
    }
    | expr '*' expr {
        $$ = $1 * $3;
    }
    | expr '/' expr {
        if ($3 == 0) {
            printf("Error: Division by zero!\n");
            $$ = 0;
        }
        else {
            $$ = $1 / $3;
        }
    }
    | '-' expr %prec UMINUS {
        $$ = -$2;
    }
    | '(' expr ')' {
        $$ = $2;
    }
    ;

%%

int main() {
    printf("Enter expressions (Ctrl+D to exit):\n");
    yyparse();
    return 0;
}

int yyerror(const char *s) {
    printf("Error: %s\n", s);
    return 0;
}




yacc -d calc.y
flex calc.l
gcc y.tab.c lex.yy.c -o calc -lfl
./calc
