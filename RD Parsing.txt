//parse1

#include <stdio.h>

int l = 0;

void A(char c[]) {
    if (c[l] == 'a')
        l++;
}

void S(char c[]) {
    A(c);

    if (c[l] == 'b')
        l++;
    else
        return;
}

int main() {
    char c[100];

    printf("Enter string: ");
    scanf("%99s", c);

    S(c);

    if (c[l] == '$')
        printf("Parsing successful");
    else
        printf("Parsing unsuccessful");

    return 0;
}



//parse2

#include <stdio.h>

int l = 0;

void A(char c[]) {
    if (c[l] == 'a' && c[l + 1] == 'b')
        l += 2;
    else if (c[l] == 'a')
        l++;
}

void S(char c[]) {
    if (c[l] == 'c')
        l++;

    A(c);

    if (c[l] == 'd')
        l++;
}

int main() {
    char c[100];

    printf("Enter string: ");
    scanf("%99s", c);

    S(c);

    if (c[l] == '$')
        printf("Parsing successful\n");
    else
        printf("Parsing unsuccessful\n");

    return 0;
}
